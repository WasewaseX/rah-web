    12→function toast(msg, isErr) {

    21→async function api(path, body) {
    22→  const res = await fetch("/api/" + path, {
    23→    method: "POST",
    24→    headers: { "Content-Type": "application/json" },
    25→    body: JSON.stringify(body || {})
    26→  });
    27→  const data = await res.json();
    28→  if (data && data.ok === false && data.error) toast(data.error, true);
    29→  return data;
    30→}

   210→function fmtIvl(d) {
   211→  if (d == null) return "";
   212→  if (d < 1) return "10m";
   213→  if (d < 30) return d + "d";
   214→  if (d < 365) return Math.round(d / 30) + "mo";
   215→  return Math.round(d / 365) + "y";

   219→  if (!CURRENT_CARD) return;

   228→    $("review-typewrap").classList.add("hidden");
   229→    $("review-revealwrap").classList.remove("hidden");

   234→    $("review-note").textContent = r.note || "";
   235→    loadPreview();
   236→    return;
   237→  }
   238→  reviewNext();
   239→}

   315→async function pollChat(task, tries, think) {
